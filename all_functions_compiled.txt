/************************************************
 * FUNCTION: send-email-notification 
 * PATH: /Users/alexanderstrarup/supabase/functions/send-email-notification/index.ts 
 ************************************************/
import { serve } from "https://deno.land/std@0.168.0/http/server.ts";
import { createClient } from "https://esm.sh/@supabase/supabase-js@2.39.3";

const RESEND_API_KEY = Deno.env.get('RESEND_API_KEY')!;
const SUPABASE_URL = Deno.env.get('SUPABASE_URL')!;
const SUPABASE_SERVICE_ROLE_KEY = Deno.env.get('SUPABASE_SERVICE_ROLE_KEY')!;

serve(async (req) => {
  try {
    // 1. Modtag data fra databasen
    const payload = await req.json();
    const table = payload.table; 
    const record = payload.record;

    // Supabase admin client (så vi kan slå modtagerens rigtige email op)
    const supabaseAdmin = createClient(SUPABASE_URL, SUPABASE_SERVICE_ROLE_KEY);

    let receiverId;
    let emailSubject = "";
    let emailHtml = "";

    // --- HVIS DET ER EN BESKED ---
    if (table === 'messages') {
      receiverId = record.receiver_id;
      
      // Undgå systemmails (hvis HoppOn selv sender en besked)
      if (!record.sender_id) return new Response("Systembesked ignoreres", { status: 200 });

      emailSubject = "Ny besked på HoppOn 💬";
      emailHtml = `
        <div style="font-family: sans-serif; text-align: center; padding: 20px; color: #0F172A;">
          <h2 style="color: #6366F1;">Du har fået en ny besked!</h2>
          <p>Nogen har skrevet til dig vedrørende din samkørsel.</p>
          <a href="https://hoppon.dk/beskeder" style="background: #0F172A; color: #fff; padding: 14px 28px; text-decoration: none; border-radius: 12px; font-weight: bold; display: inline-block; margin-top: 15px;">Læs og svar på HoppOn</a>
        </div>
      `;
    } 
    // --- HVIS NOGEN BOOKER EN TUR ---
    else if (table === 'bookings') { 
      receiverId = record.driver_id; 
      
      emailSubject = "Nyt match! Nogen vil køre med dig 🚗";
      emailHtml = `
        <div style="font-family: sans-serif; text-align: center; padding: 20px; color: #0F172A;">
          <h2 style="color: #6366F1;">Nyt sæde booket!</h2>
          <p>En passager har lige anmodet om/booket en plads på din tur.</p>
          <a href="https://hoppon.dk/profil" style="background: #0F172A; color: #fff; padding: 14px 28px; text-decoration: none; border-radius: 12px; font-weight: bold; display: inline-block; margin-top: 15px;">Gå til dine ture</a>
        </div>
      `;
    } else {
      return new Response("Ukendt tabel", { status: 400 });
    }

    // 2. Slå modtagerens email op
    const { data: { user }, error: userError } = await supabaseAdmin.auth.admin.getUserById(receiverId);
    if (userError || !user?.email) throw new Error("Kunne ikke finde brugerens email");

    // 3. Send mailen via Resend
    const res = await fetch('https://api.resend.com/emails', {
      method: 'POST',
      headers: {
        'Content-Type': 'application/json',
        'Authorization': `Bearer ${RESEND_API_KEY}`
      },
      body: JSON.stringify({
        from: 'HoppOn <hej@hoppon.dk>', // VIGTIGT: Ret til din verificerede mail
        to: user.email,
        subject: emailSubject,
        html: emailHtml
      })
    });

    const resData = await res.json();
    return new Response(JSON.stringify(resData), { status: 200 });

  } catch (error) {
    return new Response(JSON.stringify({ error: error.message }), { status: 500 });
  }
});

/************************************************
 * FUNCTION: delete-trip 
 * PATH: /Users/alexanderstrarup/supabase/functions/delete-trip/index.ts 
 ************************************************/
import { serve } from "https://deno.land/std@0.168.0/http/server.ts"
import { createClient } from "https://esm.sh/@supabase/supabase-js@2.39.3"
import Stripe from 'npm:stripe@14.14.0'

// Opsæt CORS, så din frontend kan kommunikere med funktionen
const corsHeaders = {
  'Access-Control-Allow-Origin': '*',
  'Access-Control-Allow-Headers': 'authorization, x-client-info, apikey, content-type',
}

serve(async (req) => {
  // Håndter CORS preflight requests (OPTIONS)
  if (req.method === 'OPTIONS') {
    return new Response('ok', { headers: corsHeaders })
  }

  try {
    // 1. Initialiser Stripe
    const stripe = new Stripe(Deno.env.get('STRIPE_SECRET_KEY') as string, {
      apiVersion: '2023-10-16',
      httpClient: Stripe.createFetchHttpClient(),
    })
// 2. Opret Supabase klienter
    const supabaseUrl = Deno.env.get('SUPABASE_URL')!
    const supabaseAnonKey = Deno.env.get('SUPABASE_ANON_KEY')!
    const supabaseServiceKey = Deno.env.get('SUPABASE_SERVICE_ROLE_KEY')!

    const authHeader = req.headers.get('Authorization')
    if (!authHeader) throw new Error('Manglende Authorization header')

    // Standard klient til at validere brugeren
    const supabaseClient = createClient(supabaseUrl, supabaseAnonKey)
    
    // Admin klient til at overskrive databasen
    const supabaseAdmin = createClient(supabaseUrl, supabaseServiceKey)

    // 3. Hent brugeren for at sikre, at de er logget ind
    // Vi fjerner "Bearer " for kun at have selve JWT-strengen
    const jwt = authHeader.replace('Bearer ', '') 
    
    // Vi sender JWT'en direkte ind i getUser() for at tvinge validering
    const { data: { user }, error: userError } = await supabaseClient.auth.getUser(jwt)
    
    if (userError || !user) {
      console.error("Detaljeret Auth Fejl:", userError) // Logger den præcise fejl fra Supabase
      throw new Error('Ikke autoriseret')
    }
    // 4. Hent trip_id fra request body
    const { trip_id } = await req.json()
    if (!trip_id) throw new Error('trip_id mangler i request')

    // 5. Verificer at brugeren rent faktisk ejer turen (er chaufføren) og at den ikke allerede er afsluttet
    const { data: ride, error: rideError } = await supabaseAdmin
      .from('rides')
      .select('driver_id, status')
      .eq('id', trip_id)
      .single()

    if (rideError || !ride) throw new Error('Turen blev ikke fundet')
    if (ride.driver_id !== user.id) throw new Error('Du har ikke tilladelse til at aflyse denne tur')
    if (ride.status === 'completed') throw new Error('Turen er allerede afsluttet og kan ikke aflyses')
    if (ride.status === 'cancelled') throw new Error('Turen er allerede aflyst')

    // 6. Find alle bookinger (passagerer), der afventer eller er godkendt
    const { data: bookings, error: bookingsError } = await supabaseAdmin
      .from('bookings')
      .select('id, stripe_payment_id, status')
      .eq('ride_id', trip_id)
      .in('status', ['pending', 'approved'])

    if (bookingsError) throw new Error('Kunne ikke hente passagerer')

    // 7. Frigiv pengene i Stripe for hver passager
    if (bookings && bookings.length > 0) {
      for (const booking of bookings) {
        if (booking.stripe_payment_id) {
          try {
            // Vi annullerer PaymentIntent (dette frigiver den reserverede beløb på passagerens kort)
            await stripe.paymentIntents.cancel(booking.stripe_payment_id)
            console.log(`Betaling annulleret for booking: ${booking.id}`)
          } catch (stripeErr: any) {
            // Vi logger fejlen, men afbryder ikke hele processen – måske var betalingen allerede annulleret
            console.error(`Kunne ikke annullere Stripe betaling for ${booking.id}:`, stripeErr.message)
          }
        }
        
        // Sæt booking status til aflyst af chauffør
        await supabaseAdmin
          .from('bookings')
          .update({ status: 'cancelled_by_driver' })
          .eq('id', booking.id)
      }
    }

    // 8. Aflys selve turen (Soft delete)
    const { error: updateRideError } = await supabaseAdmin
      .from('rides')
      .update({ status: 'cancelled' })
      .eq('id', trip_id)

    if (updateRideError) throw new Error('Kunne ikke opdatere turens status')

    // Returner succes
    return new Response(
      JSON.stringify({ success: true, message: 'Turen er aflyst og pengene er refunderet' }),
      { headers: { ...corsHeaders, 'Content-Type': 'application/json' }, status: 200 }
    )

  } catch (error: any) {
    console.error('Fejl i delete-trip:', error.message)
    return new Response(
      JSON.stringify({ error: error.message }),
      { headers: { ...corsHeaders, 'Content-Type': 'application/json' }, status: 400 }
    )
  }
})

/************************************************
 * FUNCTION: settle-payment 
 * PATH: /Users/alexanderstrarup/supabase/functions/settle-payment/index.ts 
 ************************************************/
import { serve } from "https://deno.land/std@0.168.0/http/server.ts"
import { createClient } from "https://esm.sh/@supabase/supabase-js@2"
import Stripe from 'https://esm.sh/stripe@12.0.0?target=deno'

const corsHeaders = {
  'Access-Control-Allow-Origin': '*',
  'Access-Control-Allow-Headers': 'authorization, x-client-info, apikey, content-type',
}

const stripe = new Stripe(Deno.env.get('STRIPE_SECRET_KEY') as string, {
  apiVersion: '2022-11-15',
  httpClient: Stripe.createFetchHttpClient(),
})

serve(async (req) => {
  if (req.method === 'OPTIONS') return new Response('ok', { headers: corsHeaders })

  try {
    const supabase = createClient(
      Deno.env.get('SUPABASE_URL') ?? '',
      Deno.env.get('SUPABASE_SERVICE_ROLE_KEY') ?? ''
    )

    const { paymentIntentId, action, ride_id } = await req.json()

    // --- CASE 1: AUTOMATISK CRON JOB (Find alle startede ture) ---
    if (action === 'auto_capture_started_rides') {
      console.log("Running scheduled auto-capture...")
      const { data: activeRides } = await supabase
        .from('rides')
        .select('id')
        .lte('departure_time', new Date().toISOString()) // Tiden er nu eller passeret
        .eq('status', 'active')

      if (!activeRides || activeRides.length === 0) {
        return new Response(JSON.stringify({ message: "Ingen ture klar til capture" }), { headers: { ...corsHeaders, "Content-Type": "application/json" } })
      }

      for (const ride of activeRides) {
        await captureAllForRide(supabase, ride.id)
      }
      return new Response(JSON.stringify({ success: true }), { headers: { ...corsHeaders, "Content-Type": "application/json" } })
    }

    // --- CASE 2: MANUELT FRA APP (Capture alle for én specifik tur) ---
    if (action === 'capture_all_for_ride' && ride_id) {
      console.log(`Manual capture requested for ride: ${ride_id}`)
      await captureAllForRide(supabase, ride_id)
      return new Response(JSON.stringify({ success: true }), { headers: { ...corsHeaders, "Content-Type": "application/json" } })
    }

    // --- CASE 3: ENKELT HANDLING (Godkend/Afvis fra UI) ---
    if (!paymentIntentId) throw new Error("Mangler Payment Intent ID")

    if (action === 'capture') {
      await safeCapture(paymentIntentId)
    } else if (action === 'cancel') {
      await stripe.paymentIntents.cancel(paymentIntentId)
    } else {
      throw new Error("Ugyldig handling.")
    }

    return new Response(JSON.stringify({ success: true }), { headers: { ...corsHeaders, "Content-Type": "application/json" } })

  } catch (error) {
    console.error("Settle Payment Error:", error.message)
    return new Response(JSON.stringify({ error: error.message }), { 
      headers: { ...corsHeaders, "Content-Type": "application/json" },
      status: 400 
    })
  }
})

// --- HJÆLPEFUNKTIONER ---

async function captureAllForRide(supabase: any, rideId: string) {
  const { data: bookings } = await supabase
    .from('bookings')
    .select('stripe_payment_id')
    .eq('ride_id', rideId)
    .eq('status', 'approved')

  if (!bookings) return

  for (const b of bookings) {
    if (b.stripe_payment_id) {
      try {
        await safeCapture(b.stripe_payment_id)
      } catch (e) {
        console.error(`Fejl ved capture af ${b.stripe_payment_id}:`, e.message)
      }
    }
  }
}

async function safeCapture(id: string) {
  const intent = await stripe.paymentIntents.retrieve(id)
  // Kun capture hvis den er 'requires_capture' (allerede autoriseret men ikke hævet)
  if (intent.status === 'requires_capture') {
    console.log(`Capturing: ${id}`)
    return await stripe.paymentIntents.capture(id)
  }
  console.log(`Skipping capture for ${id} - status er ${intent.status}`)
}

/************************************************

import { serve } from "https://deno.land/std@0.168.0/http/server.ts"
import { createClient } from "https://esm.sh/@supabase/supabase-js@2"
import Stripe from "https://esm.sh/stripe@12.0.0?target=deno"

const corsHeaders = {
  'Access-Control-Allow-Origin': '*',
  'Access-Control-Allow-Headers': 'authorization, x-client-info, apikey, content-type',
}

const stripe = new Stripe(Deno.env.get('STRIPE_SECRET_KEY') || '', {
  apiVersion: '2022-11-15',
  httpClient: Stripe.createFetchHttpClient(),
})

serve(async (req) => {
  if (req.method === 'OPTIONS') {
    return new Response('ok', { headers: corsHeaders })
  }

  try {
    const supabaseAdmin = createClient(
      Deno.env.get('SUPABASE_URL') ?? '',
      Deno.env.get('SUPABASE_SERVICE_ROLE_KEY') ?? ''
    )

    const { user_id, email, dob_day, dob_month, dob_year, client_ip } = await req.json()

    // Hent dekrypteret data fra databasen
    const { data: taxData, error: dbError } = await supabaseAdmin
      .rpc('get_decrypted_tax_info', { target_user_id: user_id })
      .single()

    if (dbError || !taxData) {
      throw new Error(`Database fejl: ${dbError?.message || 'Ingen data fundet'}`)
    }

    // --- LOGIK TIL PARSING AF ADRESSE ---
    // Forventet format fra Mapbox: "Silkeborgvej 52, 8000 Aarhus, Danmark"
    const fullAddress = taxData.address || "";
    
    // Vi leder efter 4 cifre efterfulgt af et bynavn (før det sidste komma)
    // Regex forklaret: (\d{4}) finder 4 tal, \s+ (mellemrum), ([^,]+) finder alt indtil næste komma
    const addressMatch = fullAddress.match(/(\d{4})\s+([^,]+)/);
    
    const postalCode = addressMatch ? addressMatch[1] : '1000'; // Default hvis fejl
    const city = addressMatch ? addressMatch[2].trim() : 'København';
    const streetAddress = fullAddress.split(',')[0].trim(); // Tager alt før første komma

    // Navnehåndtering
    const names = taxData.full_name.trim().split(/\s+/)
    const firstName = names[0]
    const lastName = names.length > 1 ? names.slice(1).join(' ') : 'Bruger'
    
    // Rens IBAN
    const ibanRaw = taxData.bank_iban.replace(/\s+/g, '');

    console.log(`Opretter Stripe konto for ${email}. Adresse parsed: ${streetAddress}, ${postalCode} ${city}`);

    const account = await stripe.accounts.create({
      type: 'custom',
      country: 'DK',
      email: email,
      capabilities: {
        transfers: { requested: true },
      },
      business_type: 'individual',
      business_profile: {
        mcc: '4121', // Taxicabs/Limousines
        url: 'https://hoppon.dk',
        product_description: 'Ridesharing services via HoppOn app',
      },
      individual: {
        first_name: firstName,
        last_name: lastName,
        email: email,
        dob: {
          day: Number(dob_day),
          month: Number(dob_month),
          year: Number(dob_year),
        },
        address: {
          line1: streetAddress, // Silkeborgvej 52
          city: city,          // Aarhus
          postal_code: postalCode, // 8000
          country: 'DK',
        },
        id_number: taxData.cpr, 
      },
      external_account: {
        object: 'bank_account',
        country: 'DK',
        currency: 'dkk',
        account_number: ibanRaw,
      },
      settings: {
        payouts: {
          schedule: {
            interval: 'manual', 
          },
        },
      },
      tos_acceptance: {
        date: Math.floor(Date.now() / 1000),
        ip: client_ip || "127.0.0.1",
      },
    })

    // Gem Stripe ID i user_stripe_data
    const { error: saveError } = await supabaseAdmin
      .from('user_stripe_data')
      .upsert({
        user_id: user_id,
        stripe_account_id: account.id,
        onboarding_completed: true,
        updated_at: new Date().toISOString()
      })

    if (saveError) throw new Error("Fejl ved gemning af Stripe ID: " + saveError.message)

    return new Response(JSON.stringify({ success: true, accountId: account.id }), {
      headers: { ...corsHeaders, "Content-Type": "application/json" },
      status: 200,
    })

  } catch (error) {
    console.error("Stripe Oprettelse Fejl:", error.message)
    return new Response(JSON.stringify({ error: error.message }), {
      headers: { ...corsHeaders, "Content-Type": "application/json" },
      status: 400,
    })
  }
})

/************************************************
 * FUNCTION: payment-sheet 
 * PATH: /Users/alexanderstrarup/supabase/functions/payment-sheet/index.ts 
 ************************************************/
import { serve } from "https://deno.land/std@0.168.0/http/server.ts"
import Stripe from 'https://esm.sh/stripe@12.0.0?target=deno'

const corsHeaders = {
  'Access-Control-Allow-Origin': '*',
  'Access-Control-Allow-Headers': 'authorization, x-client-info, apikey, content-type',
}

serve(async (req) => {
  // 1. CORS Preflight - Dette skal være her for at browseren tillader kaldet
  if (req.method === 'OPTIONS') {
    return new Response('ok', { headers: corsHeaders })
  }

  try {
    // 2. Vi flytter Stripe-initialiseringen IND i serve. 
    // Det løser "Missing apiKey" fejlen, da den først læser nøglen, når kaldet sker.
    const key = Deno.env.get('STRIPE_SECRET_KEY');
    if (!key) throw new Error("STRIPE_SECRET_KEY ikke fundet");

    const stripe = new Stripe(key, {
      apiVersion: '2022-11-15',
      // VI FJERNER httpClient her, da det er det, der trigger 'runMicrotasks' fejlen på web
    })

    const { amount, currency } = await req.json()

    if (!amount || amount <= 0) {
      throw new Error("Ugyldigt beløb.");
    }

    // DIN MODEL: Vi lægger de 5 kr. til her (150 -> 155)
        // NYT: Ingen ekstra gebyrer til passageren
    const totalAmount = Math.round(Number(amount) * 100); 


    const customer = await stripe.customers.create();
    
    // Vi bruger præcis din logik fra appen
    const ephemeralKey = await stripe.ephemeralKeys.create(
      { customer: customer.id },
      { apiVersion: '2022-11-15' }
    )

    const paymentIntent = await stripe.paymentIntents.create({
      amount: totalAmount,
      currency: currency || 'dkk',
      customer: customer.id,
      payment_method_types: ['card'],
      capture_method: 'manual', 
    });

    // 3. Vi sender client_secret retur (Web Elements kalder den ofte bare 'clientSecret')
    return new Response(
      JSON.stringify({
        paymentIntent: paymentIntent.client_secret,
        clientSecret: paymentIntent.client_secret, // Tilføjet for web-kompatibilitet
        paymentIntentId: paymentIntent.id,
        ephemeralKey: ephemeralKey.secret,
        customer: customer.id,
        publishableKey: Deno.env.get('STRIPE_PUBLISHABLE_KEY'),
      }),
      { headers: { ...corsHeaders, "Content-Type": "application/json" } },
    )
  } catch (error) {
    console.error("Stripe Web Error:", error.message)
    return new Response(JSON.stringify({ error: error.message }), { 
      headers: { ...corsHeaders, "Content-Type": "application/json" },
      status: 400 
    })
  }
})

/************************************************
 * FUNCTION: expire-pending-bookings 
 * PATH: /Users/alexanderstrarup/supabase/functions/expire-pending-bookings/index.ts 
 ************************************************/
import { createClient } from 'https://esm.sh/@supabase/supabase-js@2'
import Stripe from 'https://esm.sh/stripe@14.14.0?target=deno'

Deno.serve(async (req) => {
  try {
    // 1. Opsætning
    const supabaseUrl = Deno.env.get('SUPABASE_URL') ?? ''
    const supabaseKey = Deno.env.get('SUPABASE_SERVICE_ROLE_KEY') ?? ''
    const supabase = createClient(supabaseUrl, supabaseKey)
    
    const stripe = new Stripe(Deno.env.get('STRIPE_SECRET_KEY') ?? '', {
      httpClient: Stripe.createFetchHttpClient(),
    })

    // 2. Find alle 'pending' bookinger, hvor turens afgangstidspunkt er overskredet
    const nowISO = new Date().toISOString()

    // Vi bruger inner join for kun at finde bookinger, hvor turen er kørt
    const { data: expiredBookings, error } = await supabase
      .from('bookings')
      .select('id, stripe_payment_id, rides!inner(id, departure_time)')
      .eq('status', 'pending')
      .lte('rides.departure_time', nowISO) // Afgangstidspunkt er "mindre end eller lig med" NU

    if (error) throw error

    if (!expiredBookings || expiredBookings.length === 0) {
      return new Response(JSON.stringify({ message: "Ingen udløbne bookinger fundet." }), { status: 200 })
    }

    let processedCount = 0;

    // 3. Kør igennem dem og frigiv pengene
    for (const booking of expiredBookings) {
      // Annuller reservationen hos Stripe, hvis der er et payment ID
      if (booking.stripe_payment_id) {
        try {
          await stripe.paymentIntents.cancel(booking.stripe_payment_id)
        } catch (stripeErr) {
          console.error(`Kunne ikke annullere Stripe ID ${booking.stripe_payment_id}:`, stripeErr.message)
          // Vi fortsætter alligevel – måske var den allerede annulleret
        }
      }

      // Opdater status i databasen til 'rejected' (eller 'expired', hvis du foretrækker det. 'rejected' passer dog med dit nuværende UI)
      await supabase
        .from('bookings')
        .update({ status: 'rejected' })
        .eq('id', booking.id)

      processedCount++;
    }

    return new Response(
      JSON.stringify({ success: true, processed: processedCount }), 
      { status: 200, headers: { 'Content-Type': 'application/json' } }
    )

  } catch (error) {
    console.error("Cron Job Fejl:", error.message)
    return new Response(JSON.stringify({ error: error.message }), { status: 500 })
  }
})

/************************************************
 * FUNCTION: notify-user 
 * PATH: /Users/alexanderstrarup/supabase/functions/notify-user/index.ts 
 ************************************************/
import { serve } from "https://deno.land/std@0.177.0/http/server.ts"
import { createClient } from "https://esm.sh/@supabase/supabase-js@2.7.1"
import { create, getNumericDate } from "https://deno.land/x/djwt@v3.0.1/mod.ts"

// 1. Hent hele JSON-objektet fra Secrets
const serviceAccount = JSON.parse(Deno.env.get('FIREBASE_SERVICE_ACCOUNT')!)

/**
 * Funktion til at skabe en sikker forbindelse til Google
 */
async function getAccessToken() {
  const pemHeader = "-----BEGIN PRIVATE KEY-----";
  const pemFooter = "-----END PRIVATE KEY-----";
  const pemContents = serviceAccount.private_key
    .replace(pemHeader, "")
    .replace(pemFooter, "")
    .replace(/\s/g, "");

  const binaryDerString = atob(pemContents);
  const binaryDer = new Uint8Array(binaryDerString.length);
  for (let i = 0; i < binaryDerString.length; i++) {
    binaryDer[i] = binaryDerString.charCodeAt(i);
  }

  const key = await crypto.subtle.importKey(
    "pkcs8",
    binaryDer,
    { name: "RSASSA-PKCS1-v1_5", hash: "SHA-256" },
    false,
    ["sign"]
  );

  const jwt = await create(
    { alg: "RS256", typ: "JWT" },
    {
      iss: serviceAccount.client_email,
      scope: "https://www.googleapis.com/auth/firebase.messaging",
      aud: "https://oauth2.googleapis.com/token",
      exp: getNumericDate(3600),
      iat: getNumericDate(0),
    },
    key

/************************************************

 ************************************************/
import { serve } from "https://deno.land/std@0.168.0/http/server.ts"
import { createClient } from "https://esm.sh/@supabase/supabase-js@2"

const corsHeaders = {
  'Access-Control-Allow-Origin': '*',
  'Access-Control-Allow-Headers': 'authorization, x-client-info, apikey, content-type',
}

serve(async (req) => {
  if (req.method === 'OPTIONS') return new Response('ok', { headers: corsHeaders })

  try {
    const authHeader = req.headers.get('Authorization')
    if (!authHeader) throw new Error('Ingen adgangskode fundet')

    const supabaseAdmin = createClient(
      Deno.env.get('SUPABASE_URL') ?? '',
      Deno.env.get('SUPABASE_SERVICE_ROLE_KEY') ?? ''
    )

    const token = authHeader.replace('Bearer ', '')
    const { data: { user }, error: authError } = await supabaseAdmin.auth.getUser(token)
    if (authError || !user) throw new Error('Ugyldig session')

    const { trip_id } = await req.json()

    // 2. HENT TUR-DATA (Inkl. chauffør navn)
    // Vi bruger maybeSingle for at undgå crash, og tjekker efterfølgende
    const { data: ride, error: rideError } = await supabaseAdmin
      .from('rides')
      .select('*, profiles(full_name)')
      .eq('id', trip_id)
      .maybeSingle()

    if (rideError || !ride) throw new Error("Turen blev ikke fundet")
    if (ride.status === 'completed') throw new Error("Turen er allerede afregnet")
    
    if (ride.driver_id !== user.id) {
      throw new Error("Kun chaufføren kan afslutte turen.")
    }

    // 3. HENT BOOKINGER
    const { data: bookings, error: bookingError } = await supabaseAdmin
      .from('bookings')
      .select('seats_booked, passenger_id') 
      .eq('ride_id', trip_id)
      .eq('status', 'approved');

    if (bookingError) throw new Error("Kunne ikke hente bookinger");
    
    if (!bookings || bookings.length === 0) {
       await supabaseAdmin.from('rides').update({ status: 'completed' }).eq('id', trip_id);
       return new Response(JSON.stringify({ success: true, message: "Tur afsluttet uden passagerer." }), { headers: corsHeaders });
    }

    // 4. BEREGN TOTAL SÆDER & FORBERED PASSAGER-HISTORIK
    let totalSeats = 0;
    const passengerTransactions = []; 

    bookings.forEach((b) => {
      const seats = (b.seats_booked || 1);
      totalSeats += seats;

      passengerTransactions.push({
        user_id: b.passenger_id,
        ride_id: trip_id,
        amount: -(ride.price_dkk * seats),
        type: 'ride_payment',
        status: 'completed',
        description: `Tur: ${ride.origin_city} -> ${ride.destination_city}`
      });
    });
    // 5. BEREGN INDTJENING (OPDATERET LOGIK UDEN FAST GEBYR)
    const donation_per_seat = ride.platform_donation || 0;

    // Chaufføren tjener: Deres Pris minus det de valgte at donere
    const earning_per_seat = ride.price_dkk - donation_per_seat;
    const total_driver_earnings = earning_per_seat * totalSeats;

    if (total_driver_earnings < 0) throw new Error("Indtjeningen er negativ (Tjek turprisen og donationen).");

    // 6. OPDATER WALLET
    const { error: balanceError } = await supabaseAdmin.rpc('increment_pending_balance', {
      user_id: ride.driver_id,
      amount: total_driver_earnings
    })
    if (balanceError) throw new Error(`Saldo fejl: ${balanceError.message}`)

    // 7. GEM CHAUFFØR HISTORIK
    await supabaseAdmin.from('transactions').insert({
        user_id: ride.driver_id,
        ride_id: trip_id,
        amount: total_driver_earnings,
        type: 'ride_earnings',
        status: 'pending',
        description: `Tur: ${ride.origin_city} ➝ ${ride.destination_city} (${totalSeats} sæder)`
    });

    // 8. GEM PASSAGER HISTORIK
    await supabaseAdmin.from('transactions').insert(passengerTransactions);

    // 9. LUK TUREN
    await supabaseAdmin.from('rides').update({ status: 'completed' }).eq('id', trip_id);

    // --- 10. SEND SYSTEM BESKEDER (HoppOn) ---
    const messagesToSend = [];
    const timestamp = new Date().toISOString();

    // SIKKERHED: Tjek om profiles findes, ellers brug fallback navn
    const driverName = ride.profiles?.full_name || "din chauffør";

    // Besked til Chauffør
    messagesToSend.push({
      sender_id: null, // HoppOn
      receiver_id: ride.driver_id,
      content: `Turen til ${ride.destination_city} er afsluttet! Tak for turen. Husk at anmelde dine passagerer.`,
      ride_id: trip_id,
      is_read: false,
      created_at: timestamp
    });

    // Besked til hver passager
    bookings.forEach((b) => {
      messagesToSend.push({
        sender_id: null,
        receiver_id: b.passenger_id,
        content: `Du er nu fremme i ${ride.destination_city}! Tak fordi du kørte med ${driverName}. Giv en anmeldelse nu.`,
        ride_id: trip_id, // VIGTIGT: Dette bruges af "Giv Anmeldelse" knappen
        is_read: false,
        created_at: timestamp
      });
    });

    // INSERT MED FEJL-TJEK
    const { error: msgError } = await supabaseAdmin.from('messages').insert(messagesToSend);
    
    if (msgError) {
      // Vi logger fejlen, men stopper ikke funktionen, da pengene ER overført.
      console.error("FEJL VED BESKED INSERT:", JSON.stringify(msgError));
    } else {
      console.log("Systembeskeder sendt succesfuldt.");
    }

    return new Response(JSON.stringify({ 
        success: true, 
        earnings: total_driver_earnings
    }), {
      headers: { ...corsHeaders, "Content-Type": "application/json" },
      status: 200,
    })

  } catch (error) {
    console.error("COMPLETE TRIP FEJL:", error.message);
    return new Response(JSON.stringify({ error: error.message }), {
      headers: { ...corsHeaders, "Content-Type": "application/json" },
      status: 400,
    })
  }
})

    let capturePercentage = 0.0;
    let logMessage = "Afmeldt i god tid (Gratis)";

    if (hoursUntilDeparture < 3) {
      capturePercentage = 1.0;
      logMessage = "Afmeldt < 3 timer før (100% gebyr)";
    } else if (hoursUntilDeparture < 24) {
      capturePercentage = 0.5;
      logMessage = "Afmeldt < 24 timer før (50% gebyr)";
    }

    const totalPassengerPrice = ride.price_dkk * seatsBooked;
    
    // Hent donationen
    const donationPerSeat = ride.platform_donation || 0;

    // Base earnings (Hvad de ville have tjent hvis turen blev kørt)
    const driverBaseEarnings = (ride.price_dkk - donationPerSeat) * seatsBooked;


    // --- STRIPE ---
    if (stripeId) {
      if (capturePercentage === 0) {
        console.log("Annullerer betaling fuldt ud...");
        try {
            await stripe.paymentIntents.cancel(stripeId);
        } catch (e) {
            console.log("Info: Kunne ikke annullere (måske allerede gjort):", e.message);
        }
      } else {
        const amountToCapture = Math.round(totalPassengerPrice * capturePercentage * 100);
        console.log(`Capturing ${amountToCapture} øre`);
        
        try {
            await stripe.paymentIntents.capture(stripeId, {
              amount_to_capture: amountToCapture
            });
            
            const driverShare = driverBaseEarnings * capturePercentage;

            if (driverShare > 0) {
              await supabaseAdmin.rpc('distribute_cancellation_penalty', {
                p_driver_id: ride.driver_id,
                p_amount: driverShare,
                p_ride_id: ride.id,
                p_description: `Gebyr for sen aflysning (${capturePercentage * 100}%)`
              });
            }
        } catch (e) {
             // VIGTIGT: Vi logger fejlen, men crasher IKKE. 
             // Hvis pengene allerede er trukket (dit nuværende scenarie), fortsætter vi til DB update.
             console.error("Stripe Capture Error (Fortsætter alligevel):", e.message);
        }
      }
    }

    // --- DATABASE ---
    console.log("Opdaterer booking status til cancelled_by_passenger...");
    
    const { error: updateError } = await supabaseAdmin
      .from('bookings')
      .update({ status: 'cancelled_by_passenger' })
      .eq('id', booking_id);

    if (updateError) {
      throw new Error("Kunne ikke opdatere database: " + updateError.message);
    }

    console.log("Database opdateret succesfuldt.");

    // BEMÆRK: Ingen increment_seats her, da vi ikke vil ændre bilens kapacitet.

    return new Response(JSON.stringify({ success: true, message: logMessage }), {
      headers: { ...corsHeaders, "Content-Type": "application/json" },
      status: 200,
    })

  } catch (error) {
    console.error("Critical Error:", error.message);
    return new Response(JSON.stringify({ error: error.message }), {
      headers: { ...corsHeaders, "Content-Type": "application/json" },
      status: 400,
    })
  }
})

    });

    const { amount, currency } = body;

    if (!amount || amount <= 0) {
      throw new Error("Ugyldigt beløb modtaget (skal være > 0).");
    }

    // 4. Beregn priser
    // 4. Beregn priser
    const baseAmount = Number(amount);
    
    // NYT: Ingen ekstra gebyrer til passageren
    const totalAmountDKK = baseAmount; 
    const platformFeeDKK = 0;
    const totalAmountCents = totalAmountDKK * 100; // Til øre


    // 5. Opret betaling hos Stripe
    const customer = await stripe.customers.create();
    
    const ephemeralKey = await stripe.ephemeralKeys.create(

        driver_price: baseAmount.toString(),
        service_fee: platformFeeDKK.toString(),
        total_charged: totalAmountDKK.toString()
      }
    });

    // 6. Send svar tilbage til frontend

        clientSecret: paymentIntent.client_secret,

/************************************************

 * PATH: /Users/alexanderstrarup/supabase/functions/payout/index.ts 
 ************************************************/
import { serve } from "https://deno.land/std@0.168.0/http/server.ts"
import { createClient } from 'https://esm.sh/@supabase/supabase-js@2'
import Stripe from 'https://esm.sh/stripe@12.0.0?target=deno'

const corsHeaders = {
  'Access-Control-Allow-Origin': '*',
  'Access-Control-Allow-Headers': 'authorization, x-client-info, apikey, content-type',
}

const stripe = new Stripe(Deno.env.get('STRIPE_SECRET_KEY') as string, {
  apiVersion: '2022-11-15',
  httpClient: Stripe.createFetchHttpClient(),
})

serve(async (req) => {
  // Håndter CORS
  if (req.method === 'OPTIONS') return new Response('ok', { headers: corsHeaders })

  try {
    const supabase = createClient(
      Deno.env.get('SUPABASE_URL') ?? '',
      Deno.env.get('SUPABASE_SERVICE_ROLE_KEY') ?? ''
    )

    // Henter data fra Flutter. 
    // BEMÆRK: Din Flutter kode sender 'id', så vi mapper det til userId her.
    const { id, amount } = await req.json()
    const userId = id; 

    if (!userId || !amount) {
      throw new Error("Mangler bruger ID eller beløb")
    }

    // 1. Hent chaufførens Stripe Account ID fra databasen
    const { data: stripeData, error } = await supabase
      .from('user_stripe_data') 
      .select('stripe_account_id')
      .eq('user_id', userId)
      .single()

    if (error || !stripeData?.stripe_account_id) {
      throw new Error("Ingen Stripe konto fundet på denne bruger. Har de gennemført onboarding?")
    }

    const connectedAccountId = stripeData.stripe_account_id;
    const amountInCents = Math.round(amount * 100); // Stripe bruger øre/cents

    console.log(`Starter overførsel af ${amount} DKK til konto: ${connectedAccountId}`);

    // 2. Opret en Transfer (Platform -> Chauffør)
    // Dette flytter pengene fra din platform-saldo til chaufførens Stripe-saldo.
    // Stripe udbetaler derefter automatisk til deres bankkonto (typisk dagligt).
    const transfer = await stripe.transfers.create({
      amount: amountInCents,
      currency: 'dkk',
      destination: connectedAccountId,
      description: 'Udbetaling fra HoppOn',
    });

    console.log(`✅ Udbetaling succesfuld! Transfer ID: ${transfer.id}`);

    return new Response(
      JSON.stringify({ 
        success: true, 
        transferId: transfer.id,
        message: "Pengene er overført til brugerens Stripe konto." 
      }),
      { headers: { ...corsHeaders, "Content-Type": "application/json" } },
    )

  } catch (error) {
    console.error("Udbetalingsfejl:", error.message);
    return new Response(JSON.stringify({ error: error.message }), { 
      headers: { ...corsHeaders, "Content-Type": "application/json" },
      status: 400 
    })
  }
})
